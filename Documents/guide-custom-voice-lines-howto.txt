However, because Bedrock handles custom audio assets strictly through the Resource Pack (not the Behavior Pack where your script lives), you have to set up a quick bridge configuration so the game engine recognizes your custom sound files.

Here is exactly how to set it up, configure the files, and update your code.

📂 Step 1: The Resource Pack Setup
To add your own audio recordings or voice lines, you need to place them inside your Resource Pack and map them using a sound_definitions.json file.

1. Place the Audio Files
Convert your custom voice lines into .ogg format (Minecraft Bedrock prefers .ogg over .mp3 or .wav for stability). Put them here:
RP/sounds/ambient/mob/villager/custom_line_1.ogg

2. Create sound_definitions.json
Create a file named sound_definitions.json at the root of your Resource Pack (RP/sound_definitions.json). This assigns a simple string identifier that your JavaScript code can call:

JSON
{
  "format_version": "1.14.0",
  "sound_definitions": {
    "voice.villager.panic": {
      "category": "ambient",
      "sounds": [
        "sounds/ambient/mob/villager/custom_line_1",
        "sounds/ambient/mob/villager/custom_line_2"
      ]
    },
    "voice.pillager.tactical": {
      "category": "hostile",
      "sounds": [
        "sounds/ambient/mob/pillager/tactical_1"
      ]
    }
  }
}

function sendCleanChat(entity, text, playersInRange) {
    if (!entity || !entity.isValid() || entity.typeId === "minecraft:player") return;
    if (!playersInRange) return;
    const nameTag = entity.nameTag || "Something";
    
    let color = "§7"; 
    let soundId = null; // Staging var for our audio engine

    if (entity.typeId === "minecraft:villager_v2") {
        color = "§a";
        soundId = "voice.villager.panic"; // Matches your JSON definition
    } else if (entity.typeId.includes("pillager")) {
        color = "§c";
        soundId = "voice.pillager.tactical";
    } // ... keep your other color mappings intact
    
    // 🔊 AUDIO TRIGGER LAYER
    if (soundId) {
        try {
            // Plays the sound centered at the entity's exact coordinates[cite: 4]
            entity.dimension.playSound(soundId, entity.location, {
                pitch: 1.0,  // Keep it default, or use Math.random() for pitch variation!
                volume: 1.0
            });
        } catch (e) {
            // Catch silently if the sound resource hasn't loaded yet
        }
    }

    // 💬 EXISTING CHAT ROUTER BLOCK
    const formattedMessage = `<${color}${nameTag}§r> "${text}"`;[cite: 1]
    for (const player of playersInRange) {[cite: 1]
        if (!player || !player.isValid()) continue;[cite: 1]
        try {
            if (typeof player.sendMessage === "function") {[cite: 1]
                player.sendMessage({ text: formattedMessage });[cite: 1]
            }
        } catch (e) {
            try {
                if (typeof player.sendMessage === "function") {[cite: 1]
                    player.sendMessage(formattedMessage);[cite: 1]
                }
            } catch (innerError) {}
        }
    }
}
